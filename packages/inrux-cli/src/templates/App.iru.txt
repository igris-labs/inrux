<view>
    <Column class="app">
        <Text text="Hello, Inrux" />
        <Counter label="First" :count={first} @increment={increment_first} />
        <Counter label="Second" :count={second} @increment={increment_second} />
    </Column>
</view>
<rust>
use crate::components::counter as Counter;
use crate::composables::use_counter::use_counter;
use inrux::prelude::*;

let first: Signal<u32> = use_counter();
let second: Signal<u32> = use_counter();

/// Ordinary state-changing functions need no action enum.
fn increment_first(state: &mut State, _: &Context<State>) {
    state.first.update(|n| *n += 1);
}
/// The second instance retains its own value.
fn increment_second(state: &mut State, _: &Context<State>) {
    state.second.update(|n| *n += 1);
}
</rust>
